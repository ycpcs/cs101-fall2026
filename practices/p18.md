What is the exact output?

```cpp
double computeAverage(const double array[], int size) {
    double sum = 0.0;

    }

    return sum / size;
}

int main(void) {
    double values[4] = {2.0, 4.0, 6.0, 8.0};

    return 0;
```

## Practice 4: Multidimensional Array Parameters

What is the exact output? Why must the number of columns be specified in the function parameter?

```cpp
#define COLS 3

int rowSum(int table[][COLS], int row) {
    int sum = 0;

    for (int i = 0; i < COLS; i++) {
        sum += table[row][i];
    }

    return sum;
}

int main(void) {
    int table[2][COLS] = {
        {1, 2, 3},
        {4, 5, 6}
    };

    printf("%i\n", rowSum(table, 0));
    printf("%i\n", rowSum(table, 1));

    return 0;
}
```
