## Practice 3: Moving a Player Using Composition

What is the exact output? Why do we use `&p->loc`?

<pre>
struct Location {
    int row;
    int col;
};

struct Player {
    struct Location loc;
    int hitPoints;
};

void moveLocation(struct Location *loc, int dx, int dy) {
    loc->row += dy;
    loc->col += dx;
}

void movePlayer(struct Player *p, int dx, int dy) {
    moveLocation(&p->loc, dx, dy);
}

int main(void) {
    struct Player a = { {4, 6}, 100 };

    movePlayer(&a, 3, -2);
    movePlayer(&a, -1, 4);

    printf("%i %i\n", a.loc.row, a.loc.col);

    return 0;
}
</pre>

## Practice 4: Temporary Location and Legal Movement

What is the exact output? Why is `nextLoc` used?

<pre>
struct Location {
    int row;
    int col;
};

struct Player {
    struct Location loc;
    int hitPoints;
};

void moveLocation(struct Location *loc, int dx, int dy) {
    loc->row += dy;
    loc->col += dx;
}

void movePlayer(struct Player *p, int dx, int dy) {
    struct Location nextLoc = p->loc;

    moveLocation(&nextLoc, dx, dy);

    if (nextLoc.row >= 0 && nextLoc.col >= 0) {
        p->loc = nextLoc;
    }
}

int main(void) {
    struct Player p = { {2, 3}, 100 };

    movePlayer(&p, 4, 2);
    movePlayer(&p, -10, 1);

    printf("%i %i\n", p.loc.row, p.loc.col);

    return 0;
}
</pre>
